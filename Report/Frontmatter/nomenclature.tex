\section*{Mathematical Symbols}
\noindent
\begin{tabularx}{\textwidth}{@{}l X@{}}
\toprule
\textbf{Symbol} & \textbf{Description} \\
\midrule
$\mathcal{P}$ & Set of all monitored active and passive buses in the distribution grid \\
$\mathcal{G}$ & Set of active PV inverter agents ($\mathcal{G} \subset \mathcal{P}$) \\
$\mathcal{N}_g$ & Set of immediate electrical neighbor buses observed by agent $g$ \\
$V_p(t)$ & Per-unit voltage magnitude at bus $p \in \mathcal{P}$ at step $t$ [p.u.] \\
$\tilde{V}_p(t)$ & Centered and normalized bus voltage, $(V_p(t) - 1.0) / 0.05$ \\
$P_g(t),\ Q_g(t)$ & Active and reactive power injection setpoints of inverter $g$ [kW, kvar] \\
$P_{g,\text{avail}}(t)$ & Available active PV power of inverter $g$ before curtailment [kW] \\
$S_{g,\text{rated}}$ & Rated apparent power capacity of inverter $g$ [kVA] \\
$o_g(t)$ & Local and neighbor observation vector of inverter agent $g$ \\
$\tilde{a}_g(t)$ & Normalized policy action output in $[-1, 1]$ \\
$r_t$ & Scalar step-by-step reinforcement learning reward \\
$R_{\text{viol}}(t)$ & Primary network-wide voltage violation penalty (deadband MSE + $L_\infty$) \\
$R_{\text{cost}}(t)$ & Secondary operational cost on reactive power effort and curtailment \\
$w_1, w_2, \alpha, \lambda$ & Reward weighting and penalty hyperparameters \\
\bottomrule
\end{tabularx}
